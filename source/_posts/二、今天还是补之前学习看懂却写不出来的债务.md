# AI 生成
def subtract_numbers(num1, num2):
    """
    计算两个数字的差，并包含输入验证。

    参数:
    num1: 第一个数字（被减数）
    num2: 第二个数字（减数）

    返回:
    如果输入有效，返回两数之差；否则返回错误提示信息。
    """
    # 检查第一个输入是否为数字类型 (整数或浮点数)
    # isinstance(object, classinfo) 判断对象是否是指定类
    if not isinstance(num1, (int, float)):
        # 如果第一个输入不是数字，返回错误提示
        return "请输入有效数字："
    
    # 检查第二个输入是否为数字类型 (整数或浮点数)
    if not isinstance(num2, (int, float)):
        # 如果第二个输入不是数字，返回错误提示
        return "请输入有效数字："
    
    # 执行减法运算，计算 num1 减去 num2 的结果
    result = num1 - num2
    
    # 返回计算得到的差值
    return result

<summary>答案</summary>

```bash
cp record.txt text.txt      # 被复制的文件？-->  record.txt的内容被复制到text.txt中 所以record.txt是源文件
mv record.txt record1.txt   # 那个改名字了？-->  record.txt被改名为record1.txt
mv record1.txt /            # 什么意思？-->  将record1.txt移动到根目录下
echo "2026年4月29日学习记录" >> record.txt # 什么意思？-->  将"2026年4月29日学习记录"追加到record.txt的末尾
```
